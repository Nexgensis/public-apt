Nexgensis APT repository

Signed APT repository for nexgensis-cli (version 0.1.6).

curl -fsSL https://nexgensis.github.io/public-apt/nexgensis.gpg | sudo gpg --dearmor -o /usr/share/keyrings/nexgensis.gpg
echo "deb [signed-by=/usr/share/keyrings/nexgensis.gpg] https://nexgensis.github.io/public-apt stable main" | sudo tee /etc/apt/sources.list.d/nexgensis.list
sudo apt update && sudo apt install nexgensis-cli
